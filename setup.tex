% 支持通过 config.json 读取配置（零依赖，开箱即用）
% 若当前目录下存在 config.json，则自动读取并覆盖上述信息

\file_if_exist:nT { config.json }
  {
    % JSON 中的 LaTeX 命令需要重新分词；保留英文标题、姓名中的普通空格。
    \cs_new_protected:Npn \__swufe_load_json_rescan:n #1
      {
        \group_begin:
          \char_set_catcode_space:n { 32 }
          \JSONParseSetRescanValue \l_tmpa_tl \swufeJsonData {#1}
          \exp_args:NNNV \group_end: \tl_set:Nn \l_tmpa_tl \l_tmpa_tl
      }
    \cs_new_protected:Npn \__swufe_load_json_val:nn #1#2
      {
        \clist_map_inline:nn {#2}
          {
            \prop_if_in:NnT \swufeJsonData {##1}
              {
                \__swufe_load_json_rescan:n {##1}
                \tl_if_empty:NF \l_tmpa_tl
                  {
                    \keys_set:nx { sjtu / info } { #1 = { \exp_not:V \l_tmpa_tl } }
                  }
              }
          }
      }
    \cs_new_protected:Npn \__swufe_load_json_bool:nn #1#2
      {
        \clist_map_inline:nn {#2}
          {
            \prop_if_in:NnT \swufeJsonData {##1}
              {
                \__swufe_load_json_rescan:n {##1}
                \tl_if_empty:NF \l_tmpa_tl
                  {
                    \keys_set:nx { sjtu } { #1 = \tl_use:N \l_tmpa_tl }
                  }
              }
          }
      }

    \ExplSyntaxOff
    \RequirePackage{jsonparse}
    \JSONParseFromFile{\swufeJsonData}{config.json}
    \ExplSyntaxOn
